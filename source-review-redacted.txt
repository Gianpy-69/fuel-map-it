===== sql/001_schema.sql =====
-- ============================================
-- Fuel Map Italia â€” Schema Database
-- ============================================

-- Abilita PostGIS
CREATE EXTENSION IF NOT EXISTS postgis;

-- ============================================
-- 1. Anagrafica distributori
-- ============================================
CREATE TABLE distributori (
    id              SERIAL PRIMARY KEY,
    nome            VARCHAR(200) NOT NULL,
    indirizzo       VARCHAR(300),
    comune          VARCHAR(100),
    provincia       CHAR(2),
    posizione       GEOGRAPHY(POINT, 4326) NOT NULL,
    bandiera        VARCHAR(50),          -- Brand: ENI, Q8, IP, Tamoil...
    self_service    BOOLEAN DEFAULT true,
    created_at      TIMESTAMPTZ DEFAULT NOW()
);

-- Indice spaziale GIST per query veloci
CREATE INDEX idx_distributori_posizione ON distributori USING GIST (posizione);

-- Indice per ricerche per comune/provincia
CREATE INDEX idx_distributori_comune ON distributori (comune, provincia);

-- ============================================
-- 2. Stato operativo (1:1 con distributori)
-- ============================================
CREATE TABLE stato_operativo (
    id                      SERIAL PRIMARY KEY,
    distributore_id         INT NOT NULL UNIQUE REFERENCES distributori(id) ON DELETE CASCADE,
    stato                   VARCHAR(20) DEFAULT 'sconosciuto'
                            CHECK (stato IN ('aperto','chiuso','parziale','sconosciuto')),
    benzina_disponibile     BOOLEAN DEFAULT true,
    diesel_disponibile      BOOLEAN DEFAULT true,
    gpl_disponibile         BOOLEAN,
    metano_disponibile      BOOLEAN,
    prezzo_benzina          DECIMAL(4,3),   -- es. 1.879
    prezzo_diesel           DECIMAL(4,3),
    livello_coda            SMALLINT DEFAULT 0 CHECK (livello_coda BETWEEN 0 AND 3),
    conferme                INT DEFAULT 0,
    ultimo_aggiornamento    TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================
-- 3. Segnalazioni utenti (crowdsourcing)
-- ============================================
CREATE TABLE segnalazioni_utenti (
    id                  SERIAL PRIMARY KEY,
    distributore_id     INT NOT NULL REFERENCES distributori(id) ON DELETE CASCADE,
    tipo_segnalazione   VARCHAR(30) NOT NULL
                        CHECK (tipo_segnalazione IN (
                            'stato','prezzo','coda','carburante_esaurito','riaperto'
                        )),
    dettagli            JSONB DEFAULT '{}',
    session_id          VARCHAR(64),
    ip_hash             VARCHAR(64),
    posizione_utente    GEOGRAPHY(POINT, 4326),
    created_at          TIMESTAMPTZ DEFAULT NOW()
);

-- Indice per query recenti per distributore
CREATE INDEX idx_segnalazioni_distributore ON segnalazioni_utenti (distributore_id, created_at DESC);

-- Indice per rate limiting per IP
CREATE INDEX idx_segnalazioni_ip ON segnalazioni_utenti (ip_hash, created_at DESC);

const express = require('express');
const router = express.Router();
const pool = require('../db');

// GET /api/distributori/vicini?lat=41.89&lon=12.49&raggio=5000&carburante=benzina&solo_aperti=true&q=eni
router.get('/vicini', async (req, res) => {
    try {
        const lat = parseFloat(req.query.lat);
        const lon = parseFloat(req.query.lon);
        const raggio = parseInt(req.query.raggio, 10) || 5000;
        const carburante = req.query.carburante;
        const soloAperti = req.query.solo_aperti === 'true';
        const queryTesto = req.query.q ? req.query.q.trim() : null;

        if (isNaN(lat) || isNaN(lon)) {
            return res.status(400).json({ errore: 'Parametri lat e lon obbligatori' });
        }

        if (lat < 35.5 || lat > 47.1 || lon < 6.6 || lon > 18.5) {
            return res.status(400).json({ errore: 'Coordinate fuori dal territorio italiano' });
        }

        const raggioEffettivo = Math.min(raggio, 50000);

        // ESCLUDIAMO SEMPRE I DISTRIBUTORI NASCOSTI O SEGNALATI COME ERRATI
        let whereClauses = [
            `d.nascosto = false`,
            `ST_DWithin(d.posizione, ST_MakePoint($2, $1)::geography, $3)`
        ];
        let params = [lat, lon, raggioEffettivo];

        if (soloAperti) {
            whereClauses.push(`(s.stato IS NULL OR s.stato != 'chiuso')`);
        }

        if (carburante && ['benzina', 'diesel', 'gpl', 'metano'].includes(carburante.toLowerCase())) {
            whereClauses.push(`s.${carburante.toLowerCase()}_disponibile = true`);
        }

        if (queryTesto) {
            params.push(`%${queryTesto}%`);
            whereClauses.push(`(d.nome ILIKE $${params.length} OR d.bandiera ILIKE $${params.length} OR d.indirizzo ILIKE $${params.length})`);
        }

        const querySql = `
            SELECT 
                d.id,
                d.nome,
                d.indirizzo,
                d.comune,
                d.provincia,
                d.bandiera,
                d.autostradale,
                d.self_service,
                ST_Y(d.posizione::geometry) AS lat,
                ST_X(d.posizione::geometry) AS lon,
                ST_Distance(d.posizione, ST_MakePoint($2, $1)::geography) AS distanza_m,
                s.stato,
                s.benzina_disponibile,
                s.diesel_disponibile,
                s.gpl_disponibile,
                s.metano_disponibile,
                s.prezzo_benzina,
                s.prezzo_diesel,
                s.prezzo_gpl,
                s.prezzo_metano,
                s.livello_coda,
                s.conferme,
                s.ultimo_aggiornamento,
                (
                    SELECT su.dettagli->>'foto_url'
                    FROM segnalazioni_utenti su
                    WHERE su.distributore_id = d.id AND su.dettagli->>'foto_url' IS NOT NULL
                    ORDER BY su.created_at DESC
                    LIMIT 1
                ) AS ultima_foto
            FROM distributori d
            LEFT JOIN stato_operativo s ON s.distributore_id = d.id
            WHERE ${whereClauses.join(' AND ')}
            ORDER BY distanza_m ASC
            LIMIT 150
        `;

        const result = await pool.query(querySql, params);
        res.json({
            totale: result.rows.length,
            raggio_m: raggioEffettivo,
            distributori: result.rows
        });

    } catch (err) {
        console.error('Errore query distributori vicini:', err.message);
        res.status(500).json({ errore: 'Errore interno del server' });
    }
});

// POST /api/distributori/percorso â€“ cerca distributori lungo una rotta GeoJSON
router.post('/percorso', async (req, res) => {
    try {
        const { geojson, carburante, buffer_m = 2500, preferenza = 'tutte' } = req.body;

        if (!geojson || !geojson.coordinates || !Array.isArray(geojson.coordinates)) {
            return res.status(400).json({ errore: 'Tracciato GeoJSON non valido' });
        }

        const bufferEffettivo = Math.min(Math.max(parseInt(buffer_m, 10) || 2000, 500), 4000);
        let carbClause = '';
        let preferenzaClause = '';
        let orderClause = 'ORDER BY d.autostradale DESC, LEAST(s.prezzo_benzina, s.prezzo_diesel) ASC NULLS LAST';

        if (preferenza === 'solo_autostrada') {
            preferenzaClause = 'AND d.autostradale = true';
        } else if (preferenza === 'risparmio') {
            preferenzaClause = 'AND d.autostradale = false';
        }

        if (carburante && ['benzina', 'diesel', 'gpl', 'metano'].includes(carburante.toLowerCase())) {
            const c = carburante.toLowerCase();
            carbClause = `AND s.${c}_disponibile = true AND s.prezzo_${c} IS NOT NULL`;
            orderClause = (preferenza === 'solo_autostrada' || preferenza === 'risparmio')
                ? `ORDER BY s.prezzo_${c} ASC`
                : `ORDER BY d.autostradale DESC, s.prezzo_${c} ASC`;
        }

        const querySql = `
            SELECT 
                d.id,
                d.nome,
                d.indirizzo,
                d.comune,
                d.provincia,
                d.bandiera,
                d.autostradale,
                d.self_service,
                ST_Y(d.posizione::geometry) AS lat,
                ST_X(d.posizione::geometry) AS lon,
                s.stato,
                s.benzina_disponibile,
                s.diesel_disponibile,
                s.gpl_disponibile,
                s.metano_disponibile,
                s.prezzo_benzina,
                s.prezzo_diesel,
                s.prezzo_gpl,
                s.prezzo_metano,
                s.livello_coda,
                s.conferme,
                ROUND(ST_Distance(d.posizione, ST_SetSRID(ST_GeomFromGeoJSON($1), 4326)::geography)) AS dist_tracciato_m
            FROM distributori d
            JOIN stato_operativo s ON s.distributore_id = d.id
            WHERE d.nascosto = false
              AND (s.stato IS NULL OR s.stato != 'chiuso')
              AND ST_DWithin(d.posizione, ST_SetSRID(ST_GeomFromGeoJSON($1), 4326)::geography, $2)
              ${carbClause}
              ${preferenzaClause}
            ${orderClause}
            LIMIT 60
        `;

        const result = await pool.query(querySql, [JSON.stringify(geojson), bufferEffettivo]);

        res.json({
            totale: result.rows.length,
            distributori: result.rows
        });

    } catch (err) {
        console.error('Errore ricerca su percorso:', err.message);
        res.status(500).json({ errore: 'Errore durante la ricerca sul percorso' });
    }
});

// ============================================================
// Rotte segnalazioni â€“ Motore di Consenso + Posizione Errata
// ============================================================

const express = require('express');
const router = express.Router();
const crypto = require('crypto');
const path = require('path');
const multer = require('multer');
const pool = require('../db');
const rateLimiter = require('../middleware/rateLimiter');

// Configurazione Upload Foto con Multer
const storage = multer.diskStorage({
    destination: (req, file, cb) => {
        cb(null, path.join(__dirname, '..', '..', 'public', 'uploads'));
    },
    filename: (req, file, cb) => {
        const ext = path.extname(file.originalname).toLowerCase() || '.jpg';
        const uniqueName = `report_${Date.now()}_${crypto.randomBytes(4).toString('hex')}${ext}`;
        cb(null, uniqueName);
    }
});

const upload = multer({
    storage,
    limits: { fileSize: 5 * 1024 * 1024 }, // Max 5 MB
    fileFilter: (req, file, cb) => {
        const allowed = /jpeg|jpg|png|webp/;
        const mime = allowed.test(file.mimetype);
        const ext = allowed.test(path.extname(file.originalname).toLowerCase());
        if (mime && ext) return cb(null, true);
        cb(new Error('Formato immagine non supportato. Usa JPG, PNG o WebP.'));
    }
});

// POST /api/segnalazioni
router.post('/', rateLimiter, upload.single('foto'), async (req, res) => {
    const client = await pool.connect();
    try {
        let { 
            distributore_id, 
            tipo_segnalazione, 
            dettagli,
            lat_utente,
            lon_utente
        } = req.body;

        if (typeof dettagli === 'string') {
            try { dettagli = JSON.parse(dettagli); } catch (e) { dettagli = {}; }
        } else if (!dettagli) {
            dettagli = {};
        }

        distributore_id = parseInt(distributore_id, 10);
        if (!distributore_id || !tipo_segnalazione) {
            return res.status(400).json({ errore: 'distributore_id e tipo_segnalazione obbligatori' });
        }

        if (req.file) {
            dettagli.foto_url = `/uploads/${req.file.filename}`;
        }

        const distCheck = await client.query('SELECT id FROM distributori WHERE id = $1', [distributore_id]);
        if (distCheck.rows.length === 0) {
            return res.status(404).json({ errore: 'Distributore non trovato' });
        }

        const sessionId = req.headers['x-session-id'] || req.body.session_id || crypto.randomUUID();
        const ipRaw = req.ip || req.connection.remoteAddress || '0.0.0.0';
        const ipHash = crypto.createHash('sha256').update(ipRaw).digest('hex').substring(0, 16);

        await client.query('BEGIN');

        const posizioneUtente = (lat_utente && lon_utente)
            ? `ST_MakePoint(${parseFloat(lon_utente)}, ${parseFloat(lat_utente)})::geography`
            : 'NULL';

        // 1. Inserimento segnalazione
        await client.query(`
            INSERT INTO segnalazioni_utenti 
                (distributore_id, tipo_segnalazione, dettagli, session_id, ip_hash, posizione_utente)
            VALUES ($1, $2, $3, $4, $5, ${posizioneUtente})
        `, [
            distributore_id,
            tipo_segnalazione,
            JSON.stringify(dettagli),
            sessionId,
            ipHash
        ]);

        // 2. SE Ãˆ UNA SEGNALAZIONE DI POSIZIONE ERRATA
        if (tipo_segnalazione === 'posizione_errata') {
            await client.query(`
                UPDATE distributori
                SET 
                    segnalazioni_errore = segnalazioni_errore + 1,
                    nascosto = CASE WHEN segnalazioni_errore + 1 >= 3 THEN true ELSE nascosto END
                WHERE id = $1
            `, [distributore_id]);

            await client.query('COMMIT');
            return res.status(201).json({
                messaggio: 'Segnalazione di posizione errata registrata. Grazie per il supporto!'
            });
        }

        // 3. ALGORITMO DI CONSENSO NORMALE (Stato / Carburante / Coda)
        const consensusResult = await client.query(`
            SELECT 
                COUNT(*) FILTER (WHERE tipo_segnalazione IN ('riaperto') OR (tipo_segnalazione = 'stato' AND dettagli->>'stato' = 'aperto')) AS voti_aperto,
                COUNT(*) FILTER (WHERE tipo_segnalazione = 'stato' AND dettagli->>'stato' = 'chiuso') AS voti_chiuso,
                COUNT(*) FILTER (WHERE tipo_segnalazione = 'carburante_esaurito' AND dettagli->>'carburante' = 'benzina') AS voti_no_benzina,
                COUNT(*) FILTER (WHERE tipo_segnalazione = 'carburante_esaurito' AND dettagli->>'carburante' = 'diesel') AS voti_no_diesel,
                ROUND(AVG((dettagli->>'livello')::numeric) FILTER (WHERE tipo_segnalazione = 'coda')) AS media_coda,
                COUNT(*) AS totale_segnalazioni_recenti
            FROM segnalazioni_utenti
            WHERE distributore_id = $1 
              AND created_at >= NOW() - INTERVAL '90 minutes'
        `, [distributore_id]);

        const c = consensusResult.rows[0];
        const votiAperto = parseInt(c.voti_aperto, 10) || 0;
        const votiChiuso = parseInt(c.voti_chiuso, 10) || 0;
        const votiNoBenzina = parseInt(c.voti_no_benzina, 10) || 0;
        const votiNoDiesel = parseInt(c.voti_no_diesel, 10) || 0;
        const mediaCoda = c.media_coda !== null ? parseInt(c.media_coda, 10) : null;

        let nuovoStato = 'aperto';
        if (votiChiuso > votiAperto) {
            nuovoStato = 'chiuso';
        } else if (votiChiuso > 0 && votiChiuso === votiAperto) {
            nuovoStato = 'parziale';
        }

        const benzinaDisponibile = votiNoBenzina < 2;
        const dieselDisponibile = votiNoDiesel < 2;

        let updateSql = `
            UPDATE stato_operativo 
            SET 
                stato = $2,
                benzina_disponibile = $3,
                diesel_disponibile = $4,
                conferme = conferme + 1,
                ultimo_aggiornamento = NOW()
        `;
        let updateParams = [distributore_id, nuovoStato, benzinaDisponibile, dieselDisponibile];

        if (mediaCoda !== null) {
            updateSql += `, livello_coda = $${updateParams.length + 1}`;
            updateParams.push(mediaCoda);
        }

        updateSql += ` WHERE distributore_id = $1`;
        await client.query(updateSql, updateParams);

        await client.query('COMMIT');

        res.status(201).json({
            messaggio: 'Segnalazione registrata con successo',
            consenso: {
                stato: nuovoStato,
                voti_aperto: votiAperto,
                voti_chiuso: votiChiuso,
                livello_coda: mediaCoda
            },
            foto_url: dettagli.foto_url || null
        });

    } catch (err) {
        await client.query('ROLLBACK');
        console.error('Errore elaborazione segnalazione:', err.message);
        res.status(500).json({ errore: err.message || 'Errore interno del server' });
    } finally {
        if (client) client.release();
    }
});

module.exports = router;

// ============================================
// API Wrapper â€” chiamate al backend
// ============================================
const API = {
    baseUrl: '',

    async getDistributoriVicini(lat, lon, raggio = 10000, filters = {}) {
        const params = new URLSearchParams({
            lat,
            lon,
            raggio
        });

        if (filters.carburante) params.append('carburante', filters.carburante);
        if (filters.soloAperti) params.append('solo_aperti', 'true');
        if (filters.q) params.append('q', filters.q);

        const url = `${this.baseUrl}/api/distributori/vicini?${params.toString()}`;
        const res = await fetch(url);
        if (!res.ok) throw new Error(`Errore API: ${res.status}`);
        return res.json();
    },

    async getDistributore(id) {
        const res = await fetch(`${this.baseUrl}/api/distributori/${id}`);
        if (!res.ok) throw new Error(`Errore API: ${res.status}`);
        return res.json();
    },

    async getStoricoDistributore(id) {
        const res = await fetch(`${this.baseUrl}/api/distributori/${id}/storico`);
        if (!res.ok) throw new Error(`Errore API: ${res.status}`);
        return res.json();
    },

    async getDistributoriSuPercorso(geojson, carburante = 'benzina', preferenza = 'tutte') {
        const res = await fetch(`${this.baseUrl}/api/distributori/percorso`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ geojson, carburante, buffer_m: 2500, preferenza })
        });
        if (!res.ok) throw new Error(`Errore calcolo percorso: ${res.status}`);
        return res.json();
    },

    async geocode(query) {
        const url = `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query + ', Italia')}&limit=1`;
        const res = await fetch(url);
        if (!res.ok) throw new Error('Errore geocodifica cittÃ ');
        const data = await res.json();
        if (!data || data.length === 0) throw new Error(`CittÃ  "${query}" non trovata`);
        return {
            lat: parseFloat(data[0].lat),
            lon: parseFloat(data[0].lon),
            nome: data[0].display_name
        };
    },

    async getOsrmRoute(startLon, startLat, endLon, endLat) {
        const url = `https://router.project-osrm.org/route/v1/driving/${startLon},${startLat};${endLon},${endLat}?overview=full&geometries=geojson`;
        const res = await fetch(url);
        if (!res.ok) throw new Error('Impossibile tracciare il percorso stradale');
        const data = await res.json();
        if (!data.routes || data.routes.length === 0) throw new Error('Nessuna rotta stradale trovata');
        return data.routes[0];
    },

    async inviaSegnalazione(data, file = null) {
        const sessionId = this.getSessionId();
        let body;
        let headers = {
            'X-Session-Id': sessionId
        };

        if (file) {
            // Invio multipart con file
            const formData = new FormData();
            formData.append('distributore_id', data.distributore_id);
            formData.append('tipo_segnalazione', data.tipo_segnalazione);
            formData.append('dettagli', JSON.stringify(data.dettagli || {}));
            if (data.lat_utente) formData.append('lat_utente', data.lat_utente);
            if (data.lon_utente) formData.append('lon_utente', data.lon_utente);
            formData.append('foto', file);
            body = formData;
        } else {
            // Invio JSON classico
            headers['Content-Type'] = 'application/json';
            body = JSON.stringify(data);
        }

        const res = await fetch(`${this.baseUrl}/api/segnalazioni`, {
            method: 'POST',
            headers,
            body
        });
        if (!res.ok) {
            const err = await res.json();
            throw new Error(err.errore || `Errore ${res.status}`);
        }
        return res.json();
    },

    getSessionId() {
        let id = localStorage.getItem('fuelmap_session');
        if (!id) {
            id = crypto.randomUUID();
            localStorage.setItem('fuelmap_session', id);
        }
        return id;
    }
};

    async calculateRouteAndFindFuel() {
        const startInput = document.getElementById('routeStartInput');
        const destInput = document.getElementById('routeDestInput');
        const carbSelect = document.getElementById('routeCarbSelect');
        const typeSelect = document.getElementById('routeTypeSelect');
        const routeModal = document.getElementById('routeModalCard');
        const btnCalc = document.getElementById('btnCalcRoute');
        const btnClear = document.getElementById('btnClearRoute');

        const startVal = startInput.value.trim();
        const destVal = destInput.value.trim();
        const carburante = carbSelect.value;
        const preferenza = typeSelect ? typeSelect.value : 'tutte';

        if (!destVal) {
            this.showToast('âš ï¸ Inserisci una destinazione (es. Firenze)');
            destInput.focus();
            return;
        }

        btnCalc.disabled = true;
        btnCalc.innerHTML = '<span>â³ Calcolo rotta e prezzi...</span>';

        try {
            // 1. Coordinate di partenza
            let startCoord = null;
            if (startVal.toLowerCase().includes('posizione') && FuelMap.userLat && FuelMap.userLon) {
                startCoord = { lat: FuelMap.userLat, lon: FuelMap.userLon, nome: 'La mia posizione' };
            } else if (startVal) {
                startCoord = await API.geocode(startVal);
            } else if (FuelMap.userLat && FuelMap.userLon) {
                startCoord = { lat: FuelMap.userLat, lon: FuelMap.userLon, nome: 'La mia posizione' };
            } else {
                startCoord = { lat: this.currentLat, lon: this.currentLon, nome: 'Roma' };
            }

            // 2. Coordinate di arrivo
            const destCoord = await API.geocode(destVal);

            // 3. Calcola tracciato stradale OSRM
            const route = await API.getOsrmRoute(startCoord.lon, startCoord.lat, destCoord.lon, destCoord.lat);

            // 4. Disegna percorso blu sulla mappa
            FuelMap.drawRoute(route.geometry);

            // 5. Cerca distributori lungo il buffer del tracciato secondo la preferenza (Autostrada / Risparmio)
            const fuelRes = await API.getDistributoriSuPercorso(route.geometry, carburante, preferenza);
            const stations = fuelRes.distributori;

            if (!stations || stations.length === 0) {
                this.showToast('Nessun distributore aperto trovato direttamente sul tracciato');
                return;
            }

            // Attiva la modalitÃ  percorso (garantisce che escano SOLO le pompe del percorso)
            this.isRouteModeActive = true;

            // Trova la pompa piÃ¹ conveniente lungo tutto il viaggio
            let minPrice = Infinity;
            let cheapestStation = null;
            stations.forEach(d => {
                const p = parseFloat(d[`prezzo_${carburante}`] || d.prezzo_benzina || d.prezzo_diesel);
                if (!isNaN(p) && p < minPrice) {
                    minPrice = p;
                    cheapestStation = d;
                }
            });

            const cheapestId = cheapestStation ? cheapestStation.id : stations[0].id;
            this.cheapestStationId = cheapestId;

            // 6. Mostra ESCLUSIVAMENTE le pompe sul percorso
            FuelMap.renderDistributori(stations, cheapestId, carburante);

            const km = (route.distance / 1000).toFixed(0);
            const mins = (route.duration / 60).toFixed(0);
            this.showToast(`ðŸš— Percorso (${km} km, ~${mins} min): ${stations.length} pompe sul tragitto!`);

            btnClear.classList.remove('hidden');
            routeModal.classList.add('hidden');

            // 7. Fa apparire SUBITO la piÃ¹ economica del tragitto per prima
            const bestToOpen = cheapestStation || stations[0];
            if (bestToOpen) {
                setTimeout(() => {
                    this.openSheet(bestToOpen, true);
                }, 750);
            }

        } catch (err) {
            console.error('Errore percorso:', err);
            this.showToast(`âŒ ${err.message || 'Errore durante la ricerca'}`);
        } finally {
            btnCalc.disabled = false;
            btnCalc.innerHTML = '<span>ðŸ” Trova Pompe sul Percorso</span>';
        }
    },

    // --- Geolocalizzazione ---
    locateUser() {
        this.updatePill('Localizzazione in corso...', true);
        if (!navigator.geolocation) {
            this.updatePill('Geolocalizzazione non supportata');
            this.loadDefault();
            return;
        }

        navigator.geolocation.getCurrentPosition(
            (pos) => {
                const { latitude, longitude } = pos.coords;
                this.currentLat = latitude;
                this.currentLon = longitude;
                
                FuelMap.setUserPosition(latitude, longitude);
                FuelMap.flyTo(latitude, longitude, 14);
                this.loadDistributori(latitude, longitude);
            },
            (err) => {
                console.warn('Errore GPS:', err.message);
                this.updatePill('GPS non disponibile â€“ mostro Roma');
                this.loadDefault();
            },
            { enableHighAccuracy: true, timeout: 10000 }
        );
    },

    loadDefault() {
        this.currentLat = 41.9028;
        this.currentLon = 12.4964;
        this.loadDistributori(this.currentLat, this.currentLon);
    },

    // Chiamato automaticamente da Leaflet quando sposti la mappa
    onMapMoved() {
        // Se stiamo visualizzando le pompe di un percorso stradale, non sovrascriverle!
        if (this.isRouteModeActive) return;

        const { lat, lon, radius, zoom } = FuelMap.getCenterAndRadius();
        this.currentLat = lat;
        this.currentLon = lon;

        // Mostra il pulsante "Cerca in quest'area"
        const searchAreaBtn = document.getElementById('btnSearchArea');
        if (searchAreaBtn) searchAreaBtn.classList.remove('hidden');

        // Se lo zoom Ã¨ troppo distante (vista tutta Italia), non auto-caricare
        if (zoom < 9) {
            this.updatePill('ðŸ” Ingrandisci la mappa per vedere i distributori', false);
            return;
        }

        clearTimeout(this.moveDebounceTimer);
        this.moveDebounceTimer = setTimeout(() => {
            this.loadDistributori(lat, lon, Math.max(radius, 8000));
            if (searchAreaBtn) searchAreaBtn.classList.add('hidden');
        }, 500);
    },

    // --- Carica distributori ---
    async loadDistributori(lat, lon, raggio = 10000) {
        try {
            const data = await API.getDistributoriVicini(lat, lon, raggio, this.activeFilters);
            const stations = data.distributori;

            // Calcolo distributore piÃ¹ conveniente (minimo prezzo benzina/diesel)
            let minPrice = Infinity;
            let cheapestId = null;
            let totalPrice = 0;
            let countPrice = 0;

            stations.forEach(d => {
                const p = parseFloat(d.prezzo_benzina || d.prezzo_diesel);
                if (!isNaN(p) && p > 1) {
                    totalPrice += p;
                    countPrice++;
                    if (p < minPrice) {
                        minPrice = p;
                        cheapestId = d.id;
                    }
                }
            });

            this.cheapestStationId = cheapestId;
            this.currentAveragePrice = countPrice > 0 ? (totalPrice / countPrice) : null;

            // Se Ã¨ attivo il filtro "ðŸ‘‘ Economico", isola solo le stazioni top per convenienza
            let displayStations = stations;
            if (this.activeFilters.soloEconomico && this.currentAveragePrice) {
                displayStations = stations
                    .filter(d => {
                        const p = parseFloat(d.prezzo_benzina || d.prezzo_diesel);
                        return !isNaN(p) && p <= this.currentAveragePrice;
                    })
                    .sort((a, b) => {
                        const pa = parseFloat(a.prezzo_benzina || a.prezzo_diesel) || 999;
                        const pb = parseFloat(b.prezzo_benzina || b.prezzo_diesel) || 999;
                        return pa - pb;
                    });
            }

            // Renderizza con la corona d'oro al piÃ¹ economico e il prezzo corrispondente al filtro
            FuelMap.renderDistributori(displayStations, cheapestId, this.activeFilters.carburante);

            const n = stations.length;
            this.updatePill(`${n} distribut${n === 1 ? 'ore' : 'ori'} nell'area visibile`, false);
            
            setTimeout(() => {
                const statusPill = document.getElementById('statusPill');
                if (statusPill) statusPill.classList.add('hidden');
            }, 3000);

        } catch (err) {
            console.error('Errore caricamento:', err);
            this.updatePill('Errore nel caricamento');
        }
    },

    // Auto-adattamento Light/Dark Mode (Sensore di sistema o Orario 19:30-07:00)
    initAutoTheme(btnTheme) {
        const saved = localStorage.getItem('fuelmap_theme');
        let shouldBeDark = false;

        if (saved) {
            shouldBeDark = saved === 'dark';
        } else {
            // Rileva sensore tema scuro telefono/sistema
            const prefersDark = window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches;
            
            // Rileva orario notturno (tra le 19:30 e le 07:00)
            const now = new Date();
            const hour = now.getHours() + (now.getMinutes() / 60);
            const isNightTime = hour >= 19.5 || hour < 7.0;

            shouldBeDark = prefersDark || isNightTime;
        }

        if (shouldBeDark) {
            document.body.classList.add('dark-theme');
            btnTheme.classList.add('active');
            FuelMap.isDarkMode = true;
            FuelMap.updateBaseLayer();
        }

        // Ascolta variazioni del tema del telefono in tempo reale
        if (window.matchMedia) {
            window.matchMedia('(prefers-color-scheme: dark)').addEventListener('change', (e) => {
                if (!localStorage.getItem('fuelmap_theme')) {
                    const isDark = e.matches;
                    document.body.classList.toggle('dark-theme', isDark);
                    btnTheme.classList.toggle('active', isDark);
                    if (FuelMap.isDarkMode !== isDark) {
                        FuelMap.isDarkMode = isDark;
                        FuelMap.updateBaseLayer();
                    }
                }
            });
        }
    },

    async renderWeeklyChart(d) {
        const container = document.getElementById('chartSvgWrap');
        const summary = document.getElementById('chartTrendSummary');
        const daysRow = document.querySelector('.weekly-days-row');
        const currentPrice = parseFloat(d.prezzo_benzina || d.prezzo_diesel);

        if (isNaN(currentPrice) || currentPrice <= 0) {
            document.getElementById('chartSection').classList.add('hidden');
            return;
        }
        document.getElementById('chartSection').classList.remove('hidden');

        try {
            const data = await API.getStoricoDistributore(d.id);
            let prices = [];
            let days = [];

            if (data.storico && data.storico.length >= 2) {
                // Estrai i prezzi storici reali
                prices = data.storico.map(r => parseFloat(r.prezzo_benzina || r.prezzo_diesel || currentPrice));
                days = data.storico.map(r => {
                    const dt = new Date(r.data);
                    return dt.toLocaleDateString('it-IT', { weekday: 'short' }).replace('.', '');
                });
            } else {
                prices = [currentPrice, currentPrice];
                days = ['Ieri', 'Oggi'];
            }

            // Aggiorna le etichette dei giorni
            if (daysRow && days.length > 0) {
                daysRow.innerHTML = days.map(day => `<span>${day}</span>`).join('');
            }

            const min = Math.min(...prices) - 0.003;
            const max = Math.max(...prices) + 0.003;
            const range = max - min || 1;
            
            const w = 320;
            const h = 60;
            
            const points = prices.map((p, i) => {
                const x = (i / (prices.length - 1)) * (w - 20) + 10;
                const y = h - ((p - min) / range) * (h - 20) - 10;
                return `${x},${y}`;
            });

            const polylinePoints = points.join(' ');
            const diff = prices[prices.length - 1] - prices[0];
            const isDecreasing = diff <= 0;
            const strokeColor = isDecreasing ? '#10B981' : '#EF4444';
            const fillColor = isDecreasing ? 'rgba(16, 185, 129, 0.12)' : 'rgba(239, 68, 68, 0.12)';

            if (Math.abs(diff) < 0.002) {
                summary.textContent = 'âž¡ï¸ Stabile questa settimana';
                summary.style.color = '#6B7280';
            } else {
                summary.textContent = isDecreasing 
                    ? `ðŸ“‰ In discesa (${(diff * 100).toFixed(1)} cent)` 
                    : `ðŸ“ˆ In rialzo (+${(diff * 100).toFixed(1)} cent)`;
                summary.style.color = strokeColor;
            }

            const svgHtml = `
                <svg viewBox="0 0 ${w} ${h}" preserveAspectRatio="none" style="width:100%; height:100%;">
                    <polygon points="10,${h} ${polylinePoints} ${w - 10},${h}" fill="${fillColor}" />
                    <polyline points="${polylinePoints}" fill="none" stroke="${strokeColor}" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round" />
                    ${points.map((pt, idx) => {
                        const [cx, cy] = pt.split(',');
                        const val = prices[idx].toFixed(3);
                        return `
                            <g class="chart-point-group">
                                <circle cx="${cx}" cy="${cy}" r="3.5" fill="#FFFFFF" stroke="${strokeColor}" stroke-width="2" />
                            </g>
                        `;
                    }).join('')}
                </svg>
            `;
            container.innerHTML = svgHtml;

        } catch (err) {
            console.warn('Errore recupero storico prezzi:', err);
        }
    },

    async submitFullReport() {
        if (!this.currentDistributore) return;

        const btn = document.getElementById('btnSubmitReport');
        btn.disabled = true;
        btn.textContent = 'Invio in corso...';

        try {
            // Invio stato principale (aperto/chiuso o altro)
            let tipoSegnalazione = 'stato';
            let dettagli = {};

            if (this.currentReportState.stato) {
                tipoSegnalazione = 'stato';
                dettagli.stato = this.currentReportState.stato;
            } else if (this.currentReportState.benzina_esaurita) {
                tipoSegnalazione = 'carburante_esaurito';
                dettagli.carburante = 'benzina';
            } else if (this.currentReportState.diesel_esaurito) {
                tipoSegnalazione = 'carburante_esaurito';
                dettagli.carburante = 'diesel';
            } else if (this.currentReportState.coda !== null) {
                tipoSegnalazione = 'coda';
                dettagli.livello = this.currentReportState.coda;
            }

            // Includiamo tutti i dettagli opzionali nel payload
            if (this.currentReportState.coda !== null) dettagli.livello = this.currentReportState.coda;
            if (this.currentReportState.benzina_esaurita) dettagli.benzina_esaurita = true;
            if (this.currentReportState.diesel_esaurito) dettagli.diesel_esaurito = true;

            const res = await API.inviaSegnalazione({
                distributore_id: this.currentDistributore.id,
                tipo_segnalazione: tipoSegnalazione,
                dettagli,
                lat_utente: FuelMap.userLat,
                lon_utente: FuelMap.userLon
            }, this.selectedPhotoFile);

            // Vibrazione aptica su cellulare se supportata
            if (navigator.vibrate) navigator.vibrate([40, 60, 40]);

            this.showToast('âœ… Segnalazione registrata con successo!');
            this.resetReportForm();
            
            // Ricarica la scheda e la mappa
            setTimeout(() => {
                const { lat, lon, radius } = FuelMap.getCenterAndRadius();
                this.loadDistributori(lat, lon, radius);
            }, 800);

        } catch (err) {
            this.showToast(`âŒ ${err.message}`);
        } finally {
            btn.textContent = 'Conferma Segnalazione';
            this.updateSubmitButtonState();
        }
    },

const FuelMap = {

    streetLayer: null,
    satelliteLayer: null,
    satelliteLabelsLayer: null,

    isSatellite: false,
    isDarkMode: false,

    init() {
        // Layer Principale Stradale OpenStreetMap (100% stabile, zero blocchi)
        this.streetLayer = L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
            attribution: 'Â© OpenStreetMap contributors',
            maxZoom: 19
        });

        // Layer Satellite Esri World Imagery
        this.satelliteLayer = L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', {
            attribution: 'Â© Esri',
            maxZoom: 19
        });

        // Etichette strade per il satellite
        this.satelliteLabelsLayer = L.tileLayer('https://services.arcgisonline.com/ArcGIS/rest/services/Reference/World_Boundaries_and_Places/MapServer/tile/{z}/{y}/{x}', {
            maxZoom: 19
        });

        this.map = L.map('map', {
            zoomControl: false,
            attributionControl: true,
            layers: [this.streetLayer]
        }).setView([41.9028, 12.4964], 13);

        L.control.zoom({ position: 'bottomright' }).addTo(this.map);

        // Inizializzazione Gruppo Cluster ad alte prestazioni
        if (typeof L.markerClusterGroup === 'function') {
            this.clusterGroup = L.markerClusterGroup({
                maxClusterRadius: 40,
                showCoverageOnHover: false,
                spiderfyOnMaxZoom: true,
                animate: true,
                iconCreateFunction: (cluster) => {
                    const count = cluster.getChildCount();
                    let sizeClass = 'cluster-sm';
                    if (count >= 15) sizeClass = 'cluster-md';
                    if (count >= 40) sizeClass = 'cluster-lg';
                    return L.divIcon({
                        html: `<div class="fuel-cluster-icon ${sizeClass}"><span>${count}</span></div>`,
                        className: '',
                        iconSize: [38, 38],
                        iconAnchor: [19, 19]
                    });
                }
            });
            this.map.addLayer(this.clusterGroup);
        }

        this.map.on('moveend', () => {
            if (window.App && typeof window.App.onMapMoved === 'function') {
                window.App.onMapMoved();
            }
        });
    },

    getCenterAndRadius() {
        const center = this.map.getCenter();
        const bounds = this.map.getBounds();
        const radiusMeters = center.distanceTo(bounds.getNorthEast());
        const zoom = this.map.getZoom();

        return {
            lat: center.lat,
            lon: center.lng,
            radius: Math.min(Math.round(radiusMeters), 30000),
            zoom
        };
    },

    createMarkerIcon(d, isCheapest = false, fuelFilter = null) {
        let price = null;
        if (fuelFilter && d[`prezzo_${fuelFilter}`]) {
            price = d[`prezzo_${fuelFilter}`];
        } else {
            price = d.prezzo_benzina || d.prezzo_diesel || d.prezzo_gpl || d.prezzo_metano;
        }

        const priceText = (price && parseFloat(price) > 0) 
            ? `${parseFloat(price).toFixed(3)} â‚¬` 
            : 'â›½';
            
        const stato = d.stato || 'sconosciuto';
        const crownHtml = isCheapest ? '<div class="pin-crown">ðŸ‘‘</div>' : '';
        const highwayTag = d.autostradale ? '<span class="pin-highway-badge" title="Area di Servizio Autostradale">ðŸ›£ï¸</span>' : '';

        const width = d.autostradale ? 80 : 66;

        return L.divIcon({
            className: 'custom-pin-wrap',
            html: `
                <div class="fuel-price-pin ${isCheapest ? 'pin-cheapest' : ''} ${d.autostradale ? 'pin-highway' : ''} pin-status-${stato}">
                    ${crownHtml}
                    ${highwayTag}
                    <span class="pin-status-dot"></span>

                </div>
            `,
            iconSize: [width, 30],
            iconAnchor: [width / 2, 30],
            popupAnchor: [0, -32]
        });
    },

    renderDistributori(distributori, cheapestId = null, fuelFilter = null) {
        if (this.clusterGroup) {
            this.clusterGroup.clearLayers();
        } else {
            this.markers.forEach(m => this.map.removeLayer(m));
        }
        this.markers = [];
        const newMarkers = [];

        distributori.forEach(d => {
            const isCheapest = d.id === cheapestId;
            const icon = this.createMarkerIcon(d, isCheapest, fuelFilter);
            
            const marker = L.marker([d.lat, d.lon], { icon });

            marker.on('click', () => {
                App.openSheet(d, isCheapest);
            });

            marker.distributoreData = d;
            newMarkers.push(marker);
            this.markers.push(marker);
        });

        if (this.clusterGroup) {
            this.clusterGroup.addLayers(newMarkers);
        } else {
            newMarkers.forEach(m => m.addTo(this.map));
        }
    },

    setUserPosition(lat, lon) {
        this.userLat = lat;

    updateBaseLayer() {
        const mapEl = document.getElementById('map');

        if (this.isSatellite) {
            // Rimuovi mappa stradale
            if (this.map.hasLayer(this.streetLayer)) this.map.removeLayer(this.streetLayer);
            // Aggiungi satellite + etichette
            if (!this.map.hasLayer(this.satelliteLayer)) this.map.addLayer(this.satelliteLayer);
            if (!this.map.hasLayer(this.satelliteLabelsLayer)) this.map.addLayer(this.satelliteLabelsLayer);
            // Disattiva filtro notturno su satellite
            if (mapEl) mapEl.classList.remove('map-dark-tiles');
        } else {
            // Rimuovi satellite + etichette
            if (this.map.hasLayer(this.satelliteLayer)) this.map.removeLayer(this.satelliteLayer);
            if (this.map.hasLayer(this.satelliteLabelsLayer)) this.map.removeLayer(this.satelliteLabelsLayer);
            // Ripristina mappa stradale
            if (!this.map.hasLayer(this.streetLayer)) this.map.addLayer(this.streetLayer);
            // Applica o rimuovi filtro notturno alla mappa stradale
            if (mapEl) mapEl.classList.toggle('map-dark-tiles', this.isDarkMode);
        }

        setTimeout(() => this.map.invalidateSize(), 50);
    },

    toggleSatellite() {
        this.isSatellite = !this.isSatellite;
        this.updateBaseLayer();
        return this.isSatellite;
    },

    toggleDarkMode(forcedState) {
        if (typeof forcedState === 'boolean') {
            this.isDarkMode = forcedState;
        } else {
            this.isDarkMode = !this.isDarkMode;
        }
        this.updateBaseLayer();
        return this.isDarkMode;
    },

    routeLayer: null,

    drawRoute(geojson) {
        this.clearRoute();

        this.routeLayer = L.geoJSON(geojson, {
            style: {
                color: '#0066FF',
                weight: 6,
                opacity: 0.85,
                lineJoin: 'round',

        try {

            if (bounds.isValid()) {
                this.map.fitBounds(bounds, { padding: [60, 60] });
            }
        } catch (e) {

        }
    },

    clearRoute() {
        if (this.routeLayer) {
            this.map.removeLayer(this.routeLayer);
            this.routeLayer = null;
        }
    }
